\pgfplotsset{
  gcLinePlot/.style={
    title style={align=center}
    , ylabel=Speedup
    , xlabel={\#Cores}
    , grid=both
    , minor tick num=4
    , grid style={line width=.1pt, draw=gray!10}
    , major grid style={line width=.2pt,draw=gray!50}
    , ymin=0
    , ymax=24
    , xmin=1
    , xmax=24
    , height=4cm
    , width=8cm
    , penciline={jag ratio=1}
    , cycle list/Accent
    , cycle multiindex* list={
        mark list*\nextlist
        Accent\nextlist
    },
  },
}

\begin{tikzpicture}

% Siege Results
\begin{scope}
  \begin{axis}[
      gcLinePlot
      , title={Siege}
    ]
    \foreach \B in {Coins,Minimax,Mss,Queens,SumEuler,Tak,TransClos,TreeSum}{
      \pgfplotstableread[col sep=comma]{data/siege/summary_\B.csv}\datatable
      \addplot+ [decoration={name=none}] table [x expr=\thisrow{cores}, y expr=\thisrow{speedup},col sep=comma] {\datatable};
    }
  \end{axis}
\end{scope}
