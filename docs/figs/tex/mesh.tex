\newcommand{\activateDown}[2]{
  \pgfmathsetmacro\z{int( #2-1 )}
  \draw [->,bus,white]             ($(pe#1#2.south)-(0.1,0)$)--($(pe#1\z.north)-(0.1,0)$);
  \draw [->,bus,Orange,very thick] ($(pe#1#2.south)-(0.1,0)$)--($(pe#1\z.north)-(0.1,0)$);
  \draw [<-,bus,white]             ($(pe#1#2.south)+(0.1,0)$)--($(pe#1\z.north)+(0.1,0)$);
  \draw [<-,bus,Orange,very thick] ($(pe#1#2.south)+(0.1,0)$)--($(pe#1\z.north)+(0.1,0)$);
}
